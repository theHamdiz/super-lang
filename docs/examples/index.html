+AuditEvent = {
    user_id: i64 = 0
    action: str = ""
}

+fetch_user(id: i64) : User {
    User{id, name: "Amina"}

+fetch_cache(id: i64) : i64 {
    id * 10

+handle_profile(id: i64) : i64 {
    audit = chan[AuditEvent](16)
    total = 0
    scope {
        worker = spawn {
            n = 0
            for _ev in audit {
                n += 1
            }
            n
        }
        // Same-line `await spawn` — equivalent to `t = spawn …; await t`.
        u = await spawn fetch_user(id)
        c = await spawn fetch_cache(id)
        audit.send(AuditEvent{user_id: u.id, action: "profile"})
        audit.close()
        processed = await worker
        total = u.id + c + processed
    }
    total
}

+distribute(n_chunks: i64) : i64 {
    hits = shared 0
    inbox = chan[LogChunk](64)
    scope {

        }
        inbox.close()
        processed = await w1
        total = hits.load()
        total + processed
    }
}

+RigidBody = {
    x: f64 = 0.0
    y: f64 = 0.0
    vx: f64 = 0.0
    vy: f64 = 0.0
    r: f64 = 1.0
    mass: f64 = 1.0

    integrate_euler(@self, dt: f64) : f64 {
        self.x += self.vx * dt
        self.y += self.vy * dt
        self.x
    }

    impulse(self, other: RigidBody) : f64 {
        dx = self.x - other.x
        dy = self.y - other.y
        dist2 = dx * dx + dy * dy
        reach = self.r + other.r
        if dist2 <= reach * reach {
            1.0
        } else {
            0.0
        }
    }
}

    min_y: f64 = 0.0

    }

            }

    }

            vy[it] = vy[it] / sp * max_v
        }

            dy = py[j] - py[i]

+accumulate_ticks(@clock, frame_dt: f64) : i64 {

use std.core.collections
use std.diag.log

+MAX_FRAME = 64
+TOPIC_COUNT = 4
+SUBSCRIBER_COUNT = 8
+MESSAGE_COUNT = 2000

FrameKind = {
    Publish
    Subscribe
    Unsubscribe
    Ping
}

+Frame = {
    wire_len: i64 = 0
    kind: i64 = 0
    topic: i64 = 0
    sequence: i64 = 0
    payload_sum: i64 = 0
}

+encode_frame(@wire, kind: i64, topic: i64, sequence: i64, payload) : i64 {
    payload_len = payload.len()
    frame_len = payload_len + 3
    wire[0] = frame_len / 256
    wire[1] = frame_len % 256
    wire[2] = kind
    wire[3] = topic
    wire[4] = sequence / 256
    wire[5] = sequence % 256
    for i in 0..payload_len {
        wire[6 + i] = payload[i]
    }
    payload_len + 6
}

+decode_frame(wire) : Frame {
    declared = wire[0] * 256 + wire[1]
    payload_len = declared - 3
    payload_sum = 0
    for i in 0..payload_len {
        payload_sum += wire[6 + i]
    }
    Frame{
        wire_len: declared,
        kind: wire[2],
        topic: wire[3],
        sequence: wire[4] * 256 + wire[5],
        payload_sum,
    }
}

+subscribe(@subscriptions, subscriber: i64, topic: i64) {
    subscriptions[subscriber * TOPIC_COUNT + topic] = 1
}

+unsubscribe(@subscriptions, subscriber: i64, topic: i64) {
    subscriptions[subscriber * TOPIC_COUNT + topic] = 0
}

+fanout(subscriptions, topic: i64, payload_sum: i64, @mailboxes) : i64 {
    delivered = 0
    for subscriber in 0..SUBSCRIBER_COUNT {
        if subscriptions[subscriber * TOPIC_COUNT + topic] == 1 {
            mailboxes[subscriber] += payload_sum
            delivered += 1
        }
    }
    delivered
}

+run_broker(messages: i64) : i64 {
    subscriptions = [0; SUBSCRIBER_COUNT * TOPIC_COUNT]
    mailboxes = [0; SUBSCRIBER_COUNT]
    for subscriber in 0..SUBSCRIBER_COUNT {
        subscribe(@subscriptions, subscriber, subscriber % TOPIC_COUNT)
        subscribe(@subscriptions, subscriber, (subscriber + 1) % TOPIC_COUNT)
    }

    wire = [0; MAX_FRAME]
    deliveries = 0
    sequence_sum = 0
    for sequence in 0..messages {
        topic = sequence % TOPIC_COUNT
        payload = [sequence % 251, topic + 17, 9]
        _written = encode_frame(@wire, 0, topic, sequence, payload)
        frame = decode_frame(wire)
        deliveries += fanout(subscriptions, frame.topic, frame.payload_sum, @mailboxes)
        sequence_sum += frame.sequence
    }
    (mailboxes.sum()) + deliveries * 31 + sequence_sum
}

+main() {
    lg = Logger.console("tcp_broker")
    checksum = run_broker(MESSAGE_COUNT)
    @lg.info("CHECKSUM={checksum}")
}

// 16_resp_lru_cache — fixed-capacity O(1) LRU/TTL core plus RESP commands.

use std.core.str
use std.diag.log

+CAPACITY = 32
+WORKLOAD = 5000

+OP_GET = 1
+OP_SET = 2
+OP_DEL = 3
+OP_EXPIRE = 4

+RespCommand = {
    op: i64 = 0
    key: i64 = 0
    value: i64 = 0
    ttl: i64 = 0
}

+parse_command(line: str) : RespCommand {
    op = split_nth(line, " ", 0)
    key = to_i64(split_nth(line, " ", 1)) or 0
    if op == "GET" {
        return RespCommand{op: OP_GET, key}
    }
    if op == "SET" {
        value = to_i64(split_nth(line, " ", 2)) or 0
        return RespCommand{op: OP_SET, key, value}
    }
    if op == "DEL" {
        return RespCommand{op: OP_DEL, key}
    }
    if op == "EXPIRE" {
        ttl = to_i64(split_nth(line, " ", 2)) or 0
        return RespCommand{op: OP_EXPIRE, key, ttl}
    }
    RespCommand{}
}

+find_slot(keys, key: i64) : i64 {
    for i in 0..CAPACITY {
        return i if keys[i] == key
    }
    0 - 1
}

+choose_slot(keys, stamps) : i64 {
    for i in 0..CAPACITY {
        return i if keys[i] == 0
    }
    oldest = 0
    for i in 1..CAPACITY {
        if stamps[i] < stamps[oldest] {
            oldest = i
        }
    }
    oldest
}

+cache_set(@keys, @values, @expires, @stamps, key: i64, value: i64, now: i64) {
    slot = find_slot(keys, key)
    if slot < 0 {
        slot = choose_slot(keys, stamps)
    }
    keys[slot] = key
    values[slot] = value
    expires[slot] = 0
    stamps[slot] = now
}

+cache_get(@keys, values, expires, @stamps, key: i64, now: i64) : i64 {
    slot = find_slot(keys, key)
    if slot < 0 {
        return 0 - 1
    }
    if expires[slot] != 0 && expires[slot] <= now {
        keys[slot] = 0
        return 0 - 1
    }
    stamps[slot] = now
    values[slot]
}

+cache_del(@keys, key: i64) : i64 {
    slot = find_slot(keys, key)
    if slot < 0 {
        return 0
    }
    keys[slot] = 0
    1
}

+cache_expire(keys, @expires, key: i64, ttl: i64, now: i64) : i64 {
    slot = find_slot(keys, key)
    if slot < 0 {
        return 0
    }
    expires[slot] = now + ttl
    1
}

+run_cache(n: i64) : i64 {

// 18_rigidbody_sandbox — fixed-timestep SoA rigid-body integration and contacts.

use std.game.physics2d
use std.math.core

+BODY_COUNT = 96
+TICKS = 180
+DT = 1.0 / 60.0
+GRAVITY = 24.0
+WORLD_WIDTH = 120.0
+WORLD_HEIGHT = 80.0

+integrate(@x, @y, @vx, @vy, radius, dt: f64) {
    for i in 0..BODY_COUNT {
        vy[i] += GRAVITY * dt
        x[i] += vx[i] * dt
        y[i] += vy[i] * dt

        if x[i] - radius[i] < 0.0 {
            x[i] = radius[i]
            vx[i] = 0.0 - vx[i] * 0.8
        }
        if x[i] + radius[i] > WORLD_WIDTH {
            x[i] = WORLD_WIDTH - radius[i]
            vx[i] = 0.0 - vx[i] * 0.8
        }
        if y[i] + radius[i] > WORLD_HEIGHT {
            y[i] = WORLD_HEIGHT - radius[i]
            vy[i] = 0.0 - vy[i] * 0.65
        }
    }
}

+resolve_pair(@x, @y, @vx, @vy, radius, a: i64, b: i64) : i64 {
    dx = x[b] - x[a]
    dy = y[b] - y[a]
    min_distance = radius[a] + radius[b]
    distance2 = dx * dx + dy * dy
    return 0 if distance2 >= min_distance * min_distance
    return 0 if distance2 < 0.000001

    distance = distance2.sqrt()
    nx = dx / distance
    ny = dy / distance
    overlap = min_distance - distance
    x[a] -= nx * overlap * 0.5
    y[a] -= ny * overlap * 0.5
    x[b] += nx * overlap * 0.5
    y[b] += ny * overlap * 0.5

    relative = (vx[b] - vx[a]) * nx + (vy[b] - vy[a]) * ny
    if relative < 0.0 {
        impulse = (0.0 - relative) * 0.9
        vx[a] -= impulse * nx
        vy[a] -= impulse * ny
        vx[b] += impulse * nx
        vy[b] += impulse * ny
    }
    1
}

+solve_contacts(@x, @y, @vx, @vy, radius) : i64 {
    contacts = 0
    for a in 0..BODY_COUNT {
        for b in (a + 1)..BODY_COUNT {
            contacts += resolve_pair(@x, @y, @vx, @vy, radius, a, b)
        }
    }
    contacts
}

+step(@x, @y, @vx, @vy, radius) : i64 {
    integrate(@x, @y, @vx, @vy, radius, DT)
    contacts = 0
    #unroll(2)
    for 0..2 {
        contacts += solve_contacts(@x, @y, @vx, @vy, radius)
    }
    contacts
}

+ChatFrame = {
    opcode: i64 = 0
    room: i64 = 0
    sender: i64 = 0
    sequence: i64 = 0
    payload_sum: i64 = 0
}

+encode_chat(@wire, opcode: i64, room: i64, sender: i64, sequence: i64, payload) : i64 {
    payload_len = payload.len()
    wire[0] = 128 + opcode
    wire[1] = payload_len + 4
    wire[2] = room
    wire[3] = sender
    wire[4] = sequence / 256
    wire[5] = sequence % 256
    for i in 0..payload_len {
        wire[6 + i] = payload[i]
    }
    payload_len + 6
}

+decode_chat(wire) : ChatFrame {
    payload_len = wire[1] - 4
    payload_sum = 0
    for i in 0..payload_len {
        payload_sum += wire[6 + i]
    }
    ChatFrame{
        opcode: wire[0] - 128,
        room: wire[2],
        sender: wire[3],
        sequence: wire[4] * 256 + wire[5],
        payload_sum,
    }
}

+room_join(@memberships, client: i64, room: i64) {
    memberships[client * ROOM_COUNT + room] = 1
}

+room_leave(@memberships, client: i64, room: i64) {
    memberships[client * ROOM_COUNT + room] = 0
}

+publish(memberships, frame: ChatFrame, @inboxes) : i64 {
    delivered = 0
    for client in 0..CLIENT_COUNT {
        if client != frame.sender && memberships[client * ROOM_COUNT + frame.room] == 1 {
            inboxes[client] += frame.payload_sum + frame.sequence
            delivered += 1
        }
    }
    delivered
}

+seed_rooms(@memberships) {
    for client in 0..CLIENT_COUNT {
        room_join(@memberships, client, client % ROOM_COUNT)
        room_join(@memberships, client, (client + 1) % ROOM_COUNT)
    }
}

+run_chat(messages: i64) : i64 {
    memberships = [0; CLIENT_COUNT * ROOM_COUNT]
    inboxes = [0; CLIENT_COUNT]
    seed_rooms(@memberships)
    wire = [0; MAX_FRAME]

        sender = sequence % CLIENT_COUNT
        room = sequence % ROOM_COUNT
        payload = [65 + sequence % 26, 32, 97 + sequence % 26]
        _size = encode_chat(@wire, OP_TEXT, room, sender, sequence, payload)
        frame = decode_chat(wire)
        deliveries += publish(memberships, frame, @inboxes)
    }
    deliveries * 53 + (inboxes.sum())
}

        if x[player] < 0.0 { x[player] += 256.0 }
        if x[player] >= 256.0 { x[player] -= 256.0 }
        if y[player] < 0.0 { y[player] += 256.0 }

        packet[cursor + 2] = qx % 256

        packet[cursor + 4] = qy % 256
        packet[cursor + 5] = tick % 251
        cursor += 6
    }

// CLI — cut a release. The plan is a method on the ship, not a free function.
//
// Run: super test examples/22_cli_ship/

use std.cli.cmd
use std.sys.fs
use std.core.{collections, str}
use std.diag.log

+Release = {
    version: str
    channel: str

    tag(self) : str => "{self.channel}-{self.version}"

    note(self) : str => "ship {self.tag()}"
}

+Ship = {
    app: str

    command(self) : Command {
        cmd = Command.new(self.app)
        cmd = cmd.about("cut and publish a build")
        cmd = cmd.version("2.4")
        cmd = cmd.author("Super")
        cmd = cmd.flag("dry-run", "n", "print the plan and stop")
        rel = Command.new("release")
        rel = rel.about("cut a release")
        rel = rel.pos("version", true, "version tag")
        rel = rel.opt("channel", "c", "NAME", "stable", "release channel", false)
        cmd = @cmd.sub(rel)
        cmd = cmd.after_help("The tag is channel-version. Flags work before or after the command. A dry run writes nothing.")
        cmd
    }

    help_page(self) : str => self.command().help()

    plan(self, argv: _) : str {
        cmd = self.command()
        m = cmd.parse(argv) or err { return "ship: {err}" }
        if m.help || m.child().help { return self.help_page() }
        if m.show_ver || m.child().show_ver { return cmd.version_line() }
        if m.sub_name != "release" { return self.help_page() }
        child = m.child()
        fallback = "0.0.0"
        built = Release{version: child.pos(0, fallback), channel: child.get("channel")}
        mode = "live"
        if m.flag("dry-run") { mode = "dry-run" }
        "{self.app} {built.note()} {mode}"
    }

    /// Live releases write `harbor.release`. A dry run only returns the plan.
    cut(self, argv: _) : str {
        line = self.plan(argv)
        if ends_with(line, " live") == false { return line }
        body = "{line}\n"
        saved = fs_write_str("harbor.release", body) or err { return "ship: {err}" }
        if saved { return "{line}\nwrote harbor.release" }
        line
    }
}

+main() {
    lg = Logger.console("ship")
    tool = Ship{app: "harbor"}
    argv = cli_args()
    rest = []
    n = len(argv)
    for 1..n { rest = arr_push(rest, argv[it]) }
    if len(rest) == 0 {
        @lg.info(tool.help_page())
        return
    }
    @lg.info(tool.cut(rest))
}

test "a release names its own tag" {
    rel = Release{version: "1.4.0", channel: "canary"}
    assert_eq(rel.tag(), "canary-1.4.0")
    assert_eq(rel.note(), "ship canary-1.4.0")

test "the ship plans a dry run from arguments" {
    tool = Ship{app: "harbor"}
    assert_eq(tool.plan(["release", "1.4.0"]), "harbor ship stable-1.4.0 live")
    assert_eq(tool.plan(["-n", "release", "-c", "canary", "1.4.0"]), "harbor ship canary-1.4.0 dry-run")
    assert_eq(tool.plan(["release", "-n", "1.4.0"]), "harbor ship stable-1.4.0 dry-run")
    assert_eq(tool.plan(["release", "-c", "canary", "1.4.0", "-n"]), "harbor ship canary-1.4.0 dry-run")

+Service = {
    name: str
    lag_ms: i64
    load: i64

    status(self) : str {
        if self.lag_ms < 20 { return "ok" }
        if self.lag_ms < 80 { return "warn" }
        "down"
    }

    ratio(self) : f64 => self.load / 100.0

    row(self) : str => "{self.name} {self.status()} {self.lag_ms}ms"
}

+Board = {
    title: str = "ops"

    frame(self) : str {
        ink = theme("accent")
        ok = theme("ok")

        @view.clear(7, 0)
        @view.block(Rect.new(0, 0, 46, 14), " {self.title} ")
        api = Service{name: "api", lag_ms: 12, load: 80}
        worker = Service{name: "worker", lag_ms: 40, load: 55}
        store = Service{name: "db", lag_ms: 8, load: 96}
        @view.gauge(Rect.new(2, 2, 16, 1), api.ratio(), "")
        @view.print_at(20, 2, api.row(), ok.fg, 0, 0)
        @view.gauge(Rect.new(2, 3, 16, 1), worker.ratio(), "")
        @view.print_at(20, 3, worker.row(), ink.fg, 0, 1)
        @view.gauge(Rect.new(2, 4, 16, 1), store.ratio(), "")
        @view.print_at(20, 4, store.row(), ok.fg, 0, 0)
        @view.sparkline(Rect.new(2, 6, 16, 1), [2, 4, 7, 5, 8, 6, 9])
        @view.table(
            Rect.new(2, 8, 42, 4),
            ["SERVICE", "STATUS", "LAG"],
            ["api", api.status(), "12ms", "worker", worker.status(), "40ms", "db", store.status(), "8ms"],

        )
        view.plain()
    }
}

+main() {
    board = Board{}
    term = Term.open()
    if term.raw == false {
        print(board.frame())
        return

        if key.kind == "ctrl" && key.ch == "c" { quit = true }
        if quit == false { sleep_ms(40) }

    @term.close()
}

+main() {
    box = Inbox{selected: 1}
    term = Term.open()
    if term.raw == false {
        print(box.frame())
        return
    }
    quit = false
    while quit == false {
        term.write("\x1b[H")
        term.write(box.frame())
        key = term.poll()
        if key.kind == "up" || key.ch == "k" {
            if box.selected > 0 { box.selected = box.selected - 1 }
        }
        if key.kind == "down" || key.ch == "j" {
            if box.selected < 2 { box.selected = box.selected + 1 }
        }
        if key.kind == "esc" || key.ch == "q" { quit = true }
        if key.kind == "ctrl" && key.ch == "c" { quit = true }
        if quit == false { sleep_ms(40) }
    }
    @term.close()
}

// Todos. The list lives in std.db.sqlite. The window is std.gui.
//
// Run: super test examples/27_gui_todo/

use std.gui
use std.db.sqlite
use std.core.str
use std.gui.window
use std.sys.time

+TodoApp: Widget = {
    conn: SqliteConn
    filter: str = "all"
    notice: str = ""
    width: i64 = 420
    height: i64 = 480

    open() : TodoApp {
        conn = sqlite_memory()
        conn = sqlite_create(conn, "todos", ["id", "title", "done", "live"]) or conn
        TodoApp{conn, filter: "all", notice: ""}
    }

    add(@self, title: str) : i64 {
        if title == "" {
            self.notice = "Title required"
            return 0
        }
        self.conn = sqlite_insert_auto(self.conn, "todos", [title, "0", "1"]) or self.conn
        self.notice = "Saved"
        sqlite_last_insert_rowid(self.conn)
    }

    toggle(@self, id: str) : str {
        rows = sqlite_find(self.conn, "todos", "id", id)
        ensure len(rows) > 0 else { return "" }
        row = rows[0]
        next = "1"
        if row[2] == "1" { next = "0" }
        self.conn = sqlite_update(self.conn, "todos", "id", id, "done", next) or self.conn
        self.notice = ""
        next
    }

    remove(@self, id: str) : i64 {
        before = sqlite_count(self.conn, "todos")
        self.conn = sqlite_delete(self.conn, "todos", "id", id) or self.conn
        before - sqlite_count(self.conn, "todos")
    }

    set_filter(@self, name: str) : str {
        self.filter = name
        self.filter
    }

    shows(self, done: str) : bool {
        if self.filter == "open" { return done == "0" }
        if self.filter == "done" { return done == "1" }
        true
    }

    lines(self) : str {
        rows = sqlite_find(self.conn, "todos", "live", "1")
        out = ""
        for row in rows {
            if self.shows(row[2]) {
                mark = " "
                if row[2] == "1" { mark = "x" }
                out = "{out}{row[0]} [{mark}] {row[1]}\n"
            }
        }
        out
    }

    screen(self, draft: str) : Gui {
        app = Gui.new("Todos", self.width, self.height)
        root = @app.column("root")
        @app.titlebar(root, "bar", "Todos")
        @app.chrome(root, "win")
        filters = @app.row(root, "filters")
        @app.button(filters, "all", "All")
        @app.button(filters, "open", "Open")
        @app.button(filters, "done", "Done")
        @app.field(root, "title", "New todo")
        @app.write_text("title", draft)
        @app.button(root, "add", "Add")
        n = sqlite_count(self.conn, "todos")
        @app.badge(root, "count", "{n}")
        rows = sqlite_find(self.conn, "todos", "live", "1")
        for row in rows {
            if self.shows(row[2]) {
                mark = " "
                if row[2] == "1" { mark = "x" }
                @app.button(root, "row{row[0]}", "{row[0]} [{mark}] {row[1]}")
            }
        }
        if self.notice != "" { @app.toast(root, "note", self.notice) }
        @app.label(root, "hint", "Type in the field, Enter adds, click a row, Esc closes")
        app
    }

    paint(self) : str {
        app = self.screen("")
        @app.layout()
        @app.paint()
    }

    /// Open the window and edit the list until the window closes.
    run(@self) : i64 {
        opened = window_open(self.width, self.height, "Todos")
        gui_open(self.width, self.height, "Todos")
        view = self.screen("")
        draft = ""
        frames = 0
        quit = false
        while quit == false {
            gui_present(self.width, self.height, view.describe())
            click = gui_click()
            key = gui_key()
            typed = gui_value("title")
            if typed != "" { draft = typed }
            if click == "add" || key == "enter" {
                @self.add(draft)
                draft = ""
                view = self.screen("")
            }
            if click == "all" || click == "open" || click == "done" {
                @self.set_filter(click)
                view = self.screen(draft)
            }
            if starts_with(click, "row") {
                @self.toggle(strip_prefix(click, "row"))
                view = self.screen(draft)
            }
            poll = window_poll()
            if poll.quit || gui_quit() || key == "esc" || opened.headless {
                quit = true
            }
            if quit == false { sleep_ms(16) }
            frames = frames + 1
        }
        frames
    }

    hit(self, x: i64, y: i64) : bool => widget_hit(x, y, 0, 0, self.width, self.height)

    measure(self, max_w: i64) : i64 => widget_measure(max_w, self.width)
}

+main() {
    app = TodoApp.open()
    @app.add("Buy milk")
    @app.add("Ship the build")
    @app.run()
}

// Breakout. Bricks, a paddle, a sprite ball, and bounce cues.
//
// Run: super test examples/28_gui_breakout/

use std.game.fx
use std.core.str
use std.gui.window
use std.sys.time

+Breakout: Widget = {
    paddle_x: i64 = 80
    ball_x: i64 = 90
    ball_y: i64 = 100
    ball_vx: i64 = 4
    ball_vy: i64 = -4
    bricks: i64 = 8
    frame_n: i64 = 0
    sound: str = ""
    width: i64 = 240
    height: i64 = 180

    tick(@self) : str {
        self.frame_n = self.frame_n + 1
        self.ball_x = self.ball_x + self.ball_vx
        self.ball_y = self.ball_y + self.ball_vy
        self.sound = ""
        if self.ball_x < 0 || self.ball_x > 200 {
            self.ball_vx = 0 - self.ball_vx
            self.sound = "bounce"
        }
        if self.ball_y < 20 {
            self.ball_vy = 0 - self.ball_vy
            if self.sound == "" { self.sound = "bounce" }
        }
        if self.ball_y >= 140 && self.ball_x >= self.paddle_x && self.ball_x <= self.paddle_x + 40 {
            self.ball_vy = 0 - 4
            self.sound = "bounce"
        }
        if self.ball_y < 40 && self.bricks > 0 {
            self.bricks = self.bricks - 1
            self.ball_vy = 4
            self.sound = "break"
        }
        self.sound
    }

    steer(@self, key: str, mouse_x: i64) : i64 {
        if key == "left" || key == "a" { self.paddle_x = self.paddle_x - 10 }
        if key == "right" || key == "d" { self.paddle_x = self.paddle_x + 10 }
        if mouse_x > 0 { self.paddle_x = mouse_x - 20 }
        if self.paddle_x < 0 { self.paddle_x = 0 }
        if self.paddle_x > 200 { self.paddle_x = 200 }
        self.paddle_x
    }

    paint(self) : str {
        ball = Sprite{name: "ball", x: self.ball_x, y: self.ball_y, w: 8, h: 8, frame: self.frame_n % 2, frames: 2}
        brick_w = self.bricks * 12
        brick = Sprite{name: "brick", x: 8, y: 24, w: brick_w, h: 10, frame: 0, frames: 1}
        heard = ""
        if self.sound != "" {
            cue = Cue{name: self.sound, gain: 80}
            heard = cue.play()
        }
        "fill 0 0 {self.width} {self.height} 12 14 28\n{ball.paint()}{brick.paint()}fill {self.paddle_x} 150 40 8 200 200 210\n{heard}"

//! Algorithms demo: pair dispatch with `when (a, b)` and `#symmetric` mirrors.
//!
//!   super test examples/algorithms/        # runs the tests below
//!   super run examples/algorithms/         # prints the dispatch table and a matrix product
//!
//! 1. `touch` picks a collision kernel from the *pair* of shapes with a tuple `when`; the mirrored arms
//!    (`Rect, Circle` next to `Circle, Rect`) reuse the same kernel with swapped arguments.
//! 2. `scale` is marked `#symmetric`: the compiler synthesizes `scale(k, v)` from `scale(v, k)` and
//!    both orders run on VM, JIT, and C23 AOT (see `tests/dispatch_family.sr`).
//! 3. `mul` from `std.math.linalg` is the same idea at library scale: nine `(kind, kind)` arms for dense,
//!    diagonal and CSR matrices.

use std.math.linalg

/// A 2D collision shape; the position lives outside so one shape can be reused.
Shape = {
    Circle(r: f64)
    Rect(w: f64, h: f64)
    Dot

    /// Short name for tables.
    name(self) : str {
        when self {
            Shape.Circle => "circle"
            Shape.Rect => "rect"
            Shape.Dot => "dot"
        }
    }

    /// Radius of a circle (zero for other shapes).
    radius(self) : f64 {
        when self {
            Shape.Circle { r } => r
            else => 0.0
        }
    }

    /// Width of a rectangle (zero for other shapes).
    width(self) : f64 {
        when self {
            Shape.Rect { w } => w
            else => 0.0
        }
    }

    /// Height of a rectangle (zero for other shapes).
    height(self) : f64 {
        when self {
            Shape.Rect { h } => h
            else => 0.0
        }
    }
}

+circle(r: f64) : Shape => Shape.Circle(r)

+rect(w: f64, h: f64) : Shape => Shape.Rect(w, h)

+dot() : Shape => Shape.Dot

/// Distance squared between two points.
+dist2(ax: f64, ay: f64, bx: f64, by: f64) : f64 {
    dx = ax - bx
    dy = ay - by
    dx * dx + dy * dy
}

/// Circle at (cx, cy) with radius r against a circle.
+circle_circle(ra: f64, ax: f64, ay: f64, rb: f64, bx: f64, by: f64) : bool {
    reach = ra + rb
    dist2(ax, ay, bx, by) <= reach * reach
}

/// Axis-aligned rectangles centred at (ax, ay) and (bx, by).
+rect_rect(aw: f64, ah: f64, ax: f64, ay: f64, bw: f64, bh: f64, bx: f64, by: f64) : bool {
    dx = ax - bx
    if dx < 0.0 { dx = -dx }
    dy = ay - by
    if dy < 0.0 { dy = -dy }
    dx <= (aw + bw) / 2.0 && dy <= (ah + bh) / 2.0
}

/// Circle against a rectangle: the circle touches when the rectangle's closest point is within the radius.
+circle_rect(r: f64, cx: f64, cy: f64, w: f64, h: f64, rx: f64, ry: f64) : bool {
    hx = w / 2.0
    hy = h / 2.0
    nx = cx
    if nx < rx - hx { nx = rx - hx }
    if nx > rx + hx { nx = rx + hx }
    ny = cy
    if ny < ry - hy { ny = ry - hy }
    if ny > ry + hy { ny = ry + hy }
    dist2(cx, cy, nx, ny) <= r * r
}

/// A point inside a rectangle.
+dot_rect(px: f64, py: f64, w: f64, h: f64, rx: f64, ry: f64) : bool {
    dx = px - rx
    if dx < 0.0 { dx = -dx }
    dy = py - ry
    if dy < 0.0 { dy = -dy }
    dx <= w / 2.0 && dy <= h / 2.0
}

/// Do two placed shapes touch? The kernel is chosen by the pair of kinds.
+touch(a: Shape, ax: f64, ay: f64, b: Shape, bx: f64, by: f64) : bool {
    when (a, b) {
        (Shape.Circle, Shape.Circle) => { return circle_circle(a.radius(), ax, ay, b.radius(), bx, by) }
        (Shape.Rect, Shape.Rect) => { return rect_rect(a.width(), a.height(), ax, ay, b.width(), b.height(), bx, by) }
        (Shape.Circle, Shape.Rect) => { return circle_rect(a.radius(), ax, ay, b.width(), b.height(), bx, by) }
        (Shape.Rect, Shape.Circle) => { return circle_rect(b.radius(), bx, by, a.width(), a.height(), ax, ay) }
        (Shape.Dot, Shape.Circle) => { return circle_circle(0.0, ax, ay, b.radius(), bx, by) }
        (Shape.Circle, Shape.Dot) => { return circle_circle(a.radius(), ax, ay, 0.0, bx, by) }
        (Shape.Dot, Shape.Rect) => { return dot_rect(ax, ay, b.width(), b.height(), bx, by) }
        (Shape.Rect, Shape.Dot) => { return dot_rect(bx, by, a.width(), a.height(), ax, ay) }
        (Shape.Dot, Shape.Dot) => { return dist2(ax, ay, bx, by) < 0.000000001 }
    }
}
